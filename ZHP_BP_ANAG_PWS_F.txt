*&---------------------------------------------------------------------*
*& GESTIONE DINAMICA SCREEN (attiva/disattiva campi)
*&---------------------------------------------------------------------*
FORM gestisci_screen.

  IF p_dip = abap_false.
    p_dipbck = abap_false.
  ENDIF.

  IF p_dir = abap_false.
    p_dirbck = abap_false.
  ENDIF.


  IF p_uniq = abap_true AND gv_uniq_prev = abap_false.
    p_dip        = abap_false.
    p_dir        = abap_false.
    p_split      = abap_false.
    gv_uniq_prev = abap_true.
  ELSEIF p_uniq = abap_false AND gv_uniq_prev = abap_true.
    p_dip        = abap_true.
    p_dir        = abap_true.
    p_split      = abap_true.
    gv_uniq_prev = abap_false.
  ENDIF.

  IF screen-name CS 'S_RCPTA'  OR screen-name CS 'S_RCPTCC'.
    screen-input = COND #( WHEN p_mail = abap_true THEN 1 ELSE 0 ).
  ENDIF.

  LOOP AT SCREEN.

    IF p_uniq = abap_true.
      IF screen-name = 'P_SPLIT' OR
         screen-name = 'P_DIP'   OR
         screen-name = 'P_PDIP'  OR
         screen-name = 'P_DIR'   OR
***         screen-name = 'S_DIR-LOW' OR screen-name = 'S_DIR-HIGH' OR
         screen-name = 'P_PDIR'.
        screen-input = 0.
      ENDIF.
    ELSE.
      IF screen-name = 'P_SPLIT'  OR
         screen-name = 'P_DIP'    OR screen-name = 'P_DIR' OR
         screen-name = 'P_DIPBCK' OR screen-name = 'P_DIRBCK'.
        screen-input = 1.
      ENDIF.
      IF screen-name = 'P_PDIP' OR screen-name = 'P_PDIPBCK' OR
**         screen-name = 'S_DIP-LOW' OR screen-name = 'S_DIP-HIGH'.
        screen-input = COND #( WHEN p_dip = abap_true THEN 1 ELSE 0 ).
      ENDIF.
      IF screen-name = 'P_PDIR'    OR screen-name = 'P_PDIRBCK' OR
***         screen-name = 'S_DIR-LOW' OR screen-name = 'S_DIR-HIGH'.
        screen-input = COND #( WHEN p_dir = abap_true THEN 1 ELSE 0 ).
      ENDIF.
    ENDIF.

    IF screen-name = 'P_PUNIQ'.
      screen-input = COND #( WHEN p_uniq = abap_true THEN 1 ELSE 0 ).
    ENDIF.
    IF screen-name = 'P_DTINZ' OR screen-name = 'P_DTFIN'.
      screen-input = COND #( WHEN p_cess = abap_true THEN 1 ELSE 0 ).
    ENDIF.
    IF screen-name = 'S_PERNR-LOW' OR screen-name = 'S_PERNR-HIGH'.
      screen-input = COND #( WHEN p_manu = abap_true THEN 1 ELSE 0 ).
    ENDIF.
    IF screen-name = 'PATH'.
      screen-input = COND #( WHEN p_floc = abap_true THEN 1 ELSE 0 ).
    ENDIF.
    IF screen-name = 'P_FLOC'.
      screen-input = COND #( WHEN p_fsrv = abap_true THEN 0 ELSE 1 ).
    ENDIF.
    IF screen-name = 'PATH_SRV'.
      screen-input = COND #( WHEN p_fsrv = abap_true THEN 1 ELSE 0 ).
    ENDIF.
    IF screen-name = 'P_FSRV'.
      screen-input = COND #( WHEN p_floc = abap_true THEN 0 ELSE 1 ).
    ENDIF.

    MODIFY SCREEN.
  ENDLOOP.

ENDFORM.

*&---------------------------------------------------------------------*
*& CONTROLLA INPUT
*&---------------------------------------------------------------------*
FORM controlla_input.

  IF p_floc = abap_true AND p_fsrv = abap_true.
    MESSAGE e398(00) WITH 'File locale e file server non possono essere selezionati insieme'.
  ENDIF.
  IF p_pdir CS '\'.
    MESSAGE e398(00) WITH |Percorso { p_pdir } non coerente per opzione indicata|.
  ENDIF.
  IF p_pdip CS '\'.
    MESSAGE e398(00) WITH |Percorso { p_pdip } non coerente per opzione indicata|.
  ENDIF.
  IF p_puniq IS NOT INITIAL AND p_puniq CS '\'.
    MESSAGE e398(00) WITH |Percorso { p_puniq } non coerente per opzione indicata|.
  ENDIF.
  IF p_atti = abap_false AND p_cess  = abap_false AND
     p_manu = abap_false AND p_floc  = abap_false AND
     p_fsrv = abap_false.
    MESSAGE e398(00) WITH 'Indicare almeno una sorgente per i PERNR'.
  ENDIF.
  IF p_cess = abap_true.
    IF p_dtinz IS INITIAL OR p_dtfin IS INITIAL.
      MESSAGE e398(00) WITH 'Indicare data DA e data A per i cessati'.
    ENDIF.
    IF p_dtinz > p_dtfin.
      MESSAGE e398(00) WITH 'Data cessazione DA deve essere <= data A'.
    ENDIF.
  ENDIF.
  IF gv_uniq_prev = abap_false.
    IF p_dip = abap_false AND p_dir = abap_false AND p_uniq = abap_false.
      MESSAGE e398(00) WITH 'Selezionare almeno un tipo di output'.
    ENDIF.
  ENDIF.
  IF p_uniq = abap_true AND p_puniq IS INITIAL.
    MESSAGE e398(00) WITH 'Indicare il percorso per il file unico'.
  ENDIF.

  IF p_dip = abap_true AND p_pdip IS INITIAL.
    MESSAGE e398(00) WITH 'Indicare il percorso per output dipendenti'.
  ENDIF.

  IF p_dir = abap_true AND p_pdir IS INITIAL.
    MESSAGE e398(00) WITH 'Indicare il percorso per output dirigenti'.
  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*& GET CID FROM CSV (file locale PC)
*&---------------------------------------------------------------------*
FORM get_cid_from_csv  TABLES   gt_inp STRUCTURE wa_input
                        USING   f_lcl  TYPE       rlgrap-filename.

  DATA: p_file TYPE string,
        p_raw  TYPE string,
        p_tab  TYPE TABLE OF string,
        lv_cnt TYPE i.

  CHECK f_lcl IS NOT INITIAL AND p_floc = abap_true.
  p_file = f_lcl.

  CALL METHOD cl_gui_frontend_services=>gui_upload
    EXPORTING
      filename                = p_file
    CHANGING
      data_tab                = p_tab
    EXCEPTIONS
      file_open_error         = 1
      file_read_error         = 2
      no_batch                = 3
      gui_refuse_filetransfer = 4
      invalid_type            = 5
      no_authority            = 6
      unknown_error           = 7
      bad_data_format         = 8
      header_not_allowed      = 9
      separator_not_allowed   = 10
      header_too_long         = 11
      unknown_dp_error        = 12
      access_denied           = 13
      dp_out_of_memory        = 14
      disk_full               = 15
      dp_timeout              = 16
      not_supported_by_gui    = 17
      error_no_gui            = 18
      OTHERS                  = 19.

  IF sy-subrc <> 0.
    MESSAGE w398(00) WITH |Errore lettura file locale: { f_lcl }|.
    RETURN.
  ENDIF.

  LOOP AT p_tab INTO p_raw.
    CONDENSE p_raw NO-GAPS.
    CHECK p_raw IS NOT INITIAL.
    APPEND VALUE ts_input( pernr = p_raw ) TO gt_inp.
    ADD 1 TO lv_cnt.
  ENDLOOP.

  msg = | - PERNR da file locale ({ f_lcl }): { lv_cnt }|.
  WRITE: / msg.

ENDFORM.

*&---------------------------------------------------------------------*
*& GET CID FROM CSV SERVER (file su application server)
*&---------------------------------------------------------------------*
FORM get_cid_from_csv_server TABLES   gt_inp STRUCTURE wa_input
                              USING   f_srv  TYPE       pfeflnamel.

  DATA: lv_line TYPE string,
        lv_cnt  TYPE i.

  CHECK f_srv IS NOT INITIAL AND p_fsrv = abap_true.

  OPEN DATASET f_srv FOR INPUT IN TEXT MODE ENCODING UTF-8.
  IF sy-subrc <> 0.
    MESSAGE w398(00) WITH |Impossibile aprire file server: { f_srv }|.
    RETURN.
  ENDIF.

  DO.
    READ DATASET f_srv INTO lv_line.
    IF sy-subrc <> 0. EXIT. ENDIF.
    CONDENSE lv_line NO-GAPS.
    CHECK lv_line IS NOT INITIAL.
    APPEND VALUE ts_input( pernr = lv_line ) TO gt_inp.
    ADD 1 TO lv_cnt.
  ENDDO.

  CLOSE DATASET f_srv.

  msg = | - PERNR da file server ({ f_srv }): { lv_cnt }|.
  WRITE: / msg.

ENDFORM.

*&---------------------------------------------------------------------*
*& CARICA DATI (bulk load con chunking per grandi volumi)
*&---------------------------------------------------------------------*
FORM carica_dati_per_anagrafica.

  CONSTANTS: lc_chunk_size TYPE i VALUE 5000.

  DATA: lt_chunk_pernr TYPE TABLE OF ts_input,
        lv_idx_from    TYPE i VALUE 1,
        lv_idx_to      TYPE i,
        lv_tot         TYPE i.

  lv_tot = lines( s_pernr ).

  " ---- PA0001 -------------------------------------------------------
  CLEAR gt_pa0001.
  lv_idx_from = 1.
  WHILE lv_idx_from <= lv_tot.
    lv_idx_to = lv_idx_from + lc_chunk_size - 1.
    IF lv_idx_to > lv_tot. lv_idx_to = lv_tot. ENDIF.
    CLEAR lt_chunk_pernr.
    LOOP AT s_pernr INTO DATA(ls_sel3) FROM lv_idx_from TO lv_idx_to.
      APPEND VALUE ts_input( pernr = ls_sel3-low ) TO lt_chunk_pernr.
    ENDLOOP.
    IF lt_chunk_pernr IS NOT INITIAL.
      DATA: lt_tmp_0001 TYPE TABLE OF ts_pa0001.
      SELECT pernr begda bukrs werks persg persk orgeh
        INTO CORRESPONDING FIELDS OF TABLE lt_tmp_0001
        FROM pa0001
        FOR ALL ENTRIES IN lt_chunk_pernr
        WHERE pernr  = lt_chunk_pernr-pernr
          AND begda <= p_data
          AND endda >= p_data
          AND bukrs IN s_bukrs.
      APPEND LINES OF lt_tmp_0001 TO gt_pa0001.
    ENDIF.
    lv_idx_from = lv_idx_to + 1.
  ENDWHILE.
  SORT gt_pa0001 BY pernr.
  DELETE ADJACENT DUPLICATES FROM gt_pa0001 COMPARING pernr.

  " ---- PA0002 -------------------------------------------------------
  CLEAR gt_pa0002.
  lv_idx_from = 1.
  WHILE lv_idx_from <= lv_tot.
    lv_idx_to = lv_idx_from + lc_chunk_size - 1.
    IF lv_idx_to > lv_tot. lv_idx_to = lv_tot. ENDIF.
    CLEAR lt_chunk_pernr.
    LOOP AT s_pernr INTO DATA(ls_sel) FROM lv_idx_from TO lv_idx_to.
      APPEND VALUE ts_input( pernr = ls_sel-low ) TO lt_chunk_pernr.
    ENDLOOP.
    IF lt_chunk_pernr IS NOT INITIAL.
      DATA: lt_tmp_0002 TYPE TABLE OF ts_pa0002.
      SELECT pernr vorna nachn perid INTO TABLE lt_tmp_0002 FROM pa0002
        FOR ALL ENTRIES IN lt_chunk_pernr
        WHERE pernr  = lt_chunk_pernr-pernr
          AND begda <= p_data
          AND endda >= p_data.
      APPEND LINES OF lt_tmp_0002 TO gt_pa0002.
    ENDIF.
    lv_idx_from = lv_idx_to + 1.
  ENDWHILE.
  SORT gt_pa0002 BY pernr.
  DELETE ADJACENT DUPLICATES FROM gt_pa0002 COMPARING pernr.

  " ---- PA0105 -------------------------------------------------------
  CLEAR gt_pa0105.
  lv_idx_from = 1.
  WHILE lv_idx_from <= lv_tot.
    lv_idx_to = lv_idx_from + lc_chunk_size - 1.
    IF lv_idx_to > lv_tot. lv_idx_to = lv_tot. ENDIF.
    CLEAR lt_chunk_pernr.
    LOOP AT s_pernr INTO DATA(ls_sel2) FROM lv_idx_from TO lv_idx_to.
      APPEND VALUE ts_input( pernr = ls_sel2-low ) TO lt_chunk_pernr.
    ENDLOOP.
    IF lt_chunk_pernr IS NOT INITIAL.
      DATA: lt_tmp_0105 TYPE TABLE OF ts_pa0105.
      SELECT pernr zmail INTO CORRESPONDING FIELDS OF TABLE lt_tmp_0105
        FROM pa0105
        FOR ALL ENTRIES IN lt_chunk_pernr
        WHERE subty  = 'ZUPN'
          AND pernr  = lt_chunk_pernr-pernr
          AND begda <= p_data
          AND endda >= p_data.
      APPEND LINES OF lt_tmp_0105 TO gt_pa0105.
    ENDIF.
    lv_idx_from = lv_idx_to + 1.
  ENDWHILE.
  SORT gt_pa0105 BY pernr.
  DELETE ADJACENT DUPLICATES FROM gt_pa0105 COMPARING pernr.

  " ---- T001 ---------------------------------------------------------
  CLEAR gt_t001.
  DATA: lt_bukrs TYPE TABLE OF ts_t001.
  LOOP AT gt_pa0001 INTO DATA(ls_0001_b).
    APPEND VALUE ts_t001( bukrs = ls_0001_b-bukrs ) TO lt_bukrs.
  ENDLOOP.
  SORT lt_bukrs BY bukrs.
  DELETE ADJACENT DUPLICATES FROM lt_bukrs COMPARING bukrs.
  IF lt_bukrs IS NOT INITIAL.
    SELECT bukrs butxt INTO TABLE gt_t001 FROM t001
      FOR ALL ENTRIES IN lt_bukrs
      WHERE bukrs = lt_bukrs-bukrs.
  ENDIF.
  SORT gt_t001 BY bukrs.

  " ---- Testi Werks --------------------------------------------------
  CLEAR gt_werks_text.
  DATA: lt_werks_tmp TYPE TABLE OF ts_werks_text.
  LOOP AT gt_pa0001 INTO DATA(ls_0001_w).
    APPEND VALUE ts_werks_text( werks = ls_0001_w-werks ) TO lt_werks_tmp.
  ENDLOOP.
  SORT lt_werks_tmp BY werks.
  DELETE ADJACENT DUPLICATES FROM lt_werks_tmp COMPARING werks.
  LOOP AT lt_werks_tmp INTO DATA(lw).
    DATA: lv_wt TYPE t500p-name1.
    CALL FUNCTION 'HRWPC_RFC_WERKS_TEXT_GET'
      EXPORTING
        werks      = lw-werks
      IMPORTING
        werks_text = lv_wt.
    APPEND VALUE ts_werks_text( werks = lw-werks name1 = lv_wt ) TO gt_werks_text.
    CLEAR lv_wt.
  ENDLOOP.
  SORT gt_werks_text BY werks.

  " ---- Focal Point --------------------------------------------------
  CLEAR gt_fp.
  DATA: lt_orgs TYPE TABLE OF ts_fp.
  LOOP AT gt_pa0001 INTO DATA(ls_0001_o).
    APPEND VALUE ts_fp( orgeh = ls_0001_o-orgeh ) TO lt_orgs.
  ENDLOOP.
  SORT lt_orgs BY orgeh.
  DELETE ADJACENT DUPLICATES FROM lt_orgs COMPARING orgeh.
  LOOP AT lt_orgs INTO DATA(lo).
    DATA: lv_cfp   TYPE hrp9202-focal_point,
          lv_cfptx TYPE zmdit129-descr.
    SELECT SINGLE focal_point INTO lv_cfp FROM hrp9202
      WHERE plvar = '01' AND otype = 'O'
        AND objid  = lo-orgeh
        AND endda >= p_data
        AND begda <= p_data.
    IF sy-subrc = 0 AND lv_cfp IS NOT INITIAL.
      SELECT SINGLE descr INTO lv_cfptx FROM zmdit129
        WHERE cdfocalpoint = lv_cfp
          AND endda >= p_data AND begda <= p_data.
    ENDIF.
    APPEND VALUE ts_fp( orgeh = lo-orgeh cod_fp = lv_cfp desc_fp = lv_cfptx ) TO gt_fp.
    CLEAR: lv_cfp, lv_cfptx.
  ENDLOOP.
  SORT gt_fp BY orgeh.

ENDFORM.


*&---------------------------------------------------------------------*
*& ESCLDI PER PERSK
*& ora che abbiamo le info da IT0001 posso esludere da S_PAERN
*& i PERNR in base al tipo dip PERSK
*&---------------------------------------------------------------------*
FORM escludi_per_persk.

  DATA: lt_pernr_da_tenere TYPE TABLE OF ts_input.

  " Cerca in gt_pa0001 (già caricata) i PERNR
  " la cui persk rientra in s_dip o s_dir
  LOOP AT gt_pa0001 INTO DATA(ls_0001).
    IF ls_0001-persk IN s_dip OR ls_0001-persk IN s_dir.
      APPEND VALUE ts_input( pernr = ls_0001-pernr ) TO lt_pernr_da_tenere.
    ENDIF.
  ENDLOOP.

  " Ricostruisce s_pernr con i soli PERNR che hanno superato il filtro
  CLEAR s_pernr[].
  LOOP AT lt_pernr_da_tenere INTO DATA(ls_t).
    s_pernr-sign   = 'I'.
    s_pernr-option = 'EQ'.
    s_pernr-low    = ls_t-pernr.
    APPEND s_pernr.
  ENDLOOP.

ENDFORM.

FORM aggiorna_progress_bar USING p_current TYPE i
                                 p_total   TYPE i
                                 p_testo   TYPE string.

  DATA: lv_perc  TYPE i,
        lv_testo TYPE c LENGTH 200.

  CHECK p_total > 0.
  lv_perc  = ( p_current * 100 ) / p_total.
  lv_testo = | { p_testo }: { p_current } / { p_total } ({ lv_perc }%)|.

  MESSAGE s398(00) WITH lv_testo.

ENDFORM.

*& WRITE FILE (application server)
*&---------------------------------------------------------------------*
FORM write_file USING p_societa  TYPE bukrs
                      p_pathname TYPE pfeflnamel.

  DATA: lv_out_file TYPE string.

  IF p_societa = 'ALL'.
    "CONCATENATE p_pathname 'ruo_registry_' p_data '_' sy-uzeit '.TXT'
    CONCATENATE p_pathname 'ruo_registry_' p_data '_' p_time '.TXT'
           INTO lv_out_file.
  ELSE.
    "CONCATENATE p_pathname 'ruo_registry_' p_data '_' sy-uzeit '_' p_societa '.TXT'
    CONCATENATE p_pathname 'ruo_registry_' p_data '_' p_time '_' p_societa '.TXT'
           INTO lv_out_file.
  ENDIF.

  OPEN DATASET lv_out_file FOR OUTPUT IN TEXT MODE ENCODING UTF-8.
  IF sy-subrc <> 0.
    WRITE: / text-e00, sy-subrc, lv_out_file.
    EXIT.
  ENDIF.

  LOOP AT gt_tabstring INTO wa_string.
    TRANSFER wa_string-line TO lv_out_file.
    IF sy-subrc <> 0.
      WRITE: / text-e00, sy-subrc, lv_out_file.
      CLOSE DATASET lv_out_file.
      EXIT.
    ENDIF.
  ENDLOOP.

  CLOSE DATASET lv_out_file.

ENDFORM.

*&---------------------------------------------------------------------*
*& SCRIVI SCARTI - ALV con CL_SALV_TABLE (factory)
*& Funziona sia online (fullscreen interattivo) sia in background
*& (spool della lista equivalente alla display list ABAP).
*&---------------------------------------------------------------------*
FORM scrivi_scarti.

  CHECK gt_scarti IS NOT INITIAL.

  " ---- Prepara una copia tipizzata per la factory ------------------
  " CL_SALV_TABLE vuole una tabella interna con tipo dichiarato
  " staticamente, quindi usiamo gt_scarti direttamente.

  DATA: lo_alv     TYPE REF TO cl_salv_table,
        lo_columns TYPE REF TO cl_salv_columns_table,
        lo_column  TYPE REF TO cl_salv_column_table,
        lo_display TYPE REF TO cl_salv_display_settings,
        lo_funcs   TYPE REF TO cl_salv_functions_list,
        lx_msg     TYPE REF TO cx_salv_msg,
        lx_no_col  TYPE REF TO cx_salv_not_found.

  " ---- Factory -----------------------------------------------------
  TRY.
      cl_salv_table=>factory(
        IMPORTING
          r_salv_table = lo_alv
        CHANGING
          t_table      = gt_scarti ).

    CATCH cx_salv_msg INTO lx_msg.
      WRITE: / |Errore inizializzazione ALV scarti: { lx_msg->get_text( ) }|.
      RETURN.
  ENDTRY.

  " ---- Funzioni standard (solo se interattivo) ---------------------
  IF cl_salv_table=>is_offline( ) = abap_false.
    lo_funcs = lo_alv->get_functions( ).
    lo_funcs->set_all( abap_true ).
  ENDIF.

  " ---- Impostazioni display ----------------------------------------
  lo_display = lo_alv->get_display_settings( ).
  lo_display->set_list_header( 'Scarti: Errori bloccanti e Warning' ).
  lo_display->set_striped_pattern( cl_salv_display_settings=>true ).

  " ---- Colonne: etichette leggibili --------------------------------
  lo_columns = lo_alv->get_columns( ).
  lo_columns->set_optimize( abap_true ).

  TRY.
      " Colonna PERNR
      lo_column ?= lo_columns->get_column( 'PERNR' ).
      lo_column->set_short_text( 'PERNR' ).
      lo_column->set_medium_text( 'CID' ).
      lo_column->set_long_text(   'Codice Dipendente' ).

      " Colonna BUKRS
      lo_column ?= lo_columns->get_column( 'BUKRS' ).
      lo_column->set_short_text( 'Soc.' ).
      lo_column->set_medium_text( 'Società' ).
      lo_column->set_long_text(   'Codice Società' ).

      " Colonna TIPO
      lo_column ?= lo_columns->get_column( 'TIPO' ).
      lo_column->set_short_text( 'Tipo' ).
      lo_column->set_medium_text( 'E/W' ).
      lo_column->set_long_text(   'Errore / Warning' ).

      " Colonna MESSAGGIO
      lo_column ?= lo_columns->get_column( 'MESSAGGIO' ).
      lo_column->set_short_text( 'Messaggio' ).
      lo_column->set_medium_text( 'Segnalazione' ).
      lo_column->set_long_text(   'Descrizione segnalazione' ).

    CATCH cx_salv_not_found INTO lx_no_col.
      " colonna non trovata: non critico, proseguiamo
  ENDTRY.

  TRY.

*&---------------------------------------------------------------------*
*&      Form  GET_PERNR_ATTIVI_V2
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*      -->P_GT_INPUT  text
*----------------------------------------------------------------------*
FORM get_pernr_attivi_v2.

  DATA: lt_raw      TYPE STANDARD TABLE OF ts_input WITH EMPTY KEY,
        ls_raw      TYPE ts_input,
        ls_existing TYPE ts_input.

  " 1) Attivi alla data p_data
  SELECT p0000~pernr, p0000~begda, p0000~endda FROM pa0001 AS p0001
    INNER JOIN pa0000 AS p0000
      ON p0000~pernr = p0001~pernr
    WHERE p0001~bukrs IN @s_bukrs
      AND p0001~begda <= @p_data
      AND p0001~endda >= @p_data
      AND p0000~begda <= @p_data
      AND p0000~endda >= @p_data
      AND p0000~stat2 = '3'
    INTO TABLE @lt_raw.

  " 2) 1 riga per PERNR: tengo BEGDA più recente
  LOOP AT lt_raw INTO ls_raw.
    READ TABLE gt_attivi WITH TABLE KEY pernr = ls_raw-pernr INTO ls_existing.

    IF sy-subrc <> 0.
      INSERT VALUE ts_input( pernr = ls_raw-pernr
                             begda = ls_raw-begda
                             endda = ls_raw-endda ) INTO TABLE gt_attivi.
    ELSEIF ls_raw-begda > ls_existing-begda.
      ls_existing-begda = ls_raw-begda.
      ls_existing-endda = ls_raw-endda.
      MODIFY TABLE gt_attivi FROM ls_existing.
    ENDIF.
  ENDLOOP.

  " 3) Output finale
  gt_attivi = CORRESPONDING #( gt_attivi ).
ENDFORM.

FORM get_pernr_cessati_v2.

  DATA: lt_raw      TYPE STANDARD TABLE OF ts_input WITH EMPTY KEY,
        ls_raw      TYPE ts_input,
        ls_existing TYPE ts_input.

  " 1) Selezione coerente su PA0000 per PERNR/BEGDA/ENDDA
  SELECT p0000~pernr, p0000~begda, p0000~endda FROM pa0000 AS p0000
    INNER JOIN pa0001 AS p0001
      ON p0001~pernr = p0000~pernr
    WHERE p0000~begda >= @p_dtinz
      AND p0000~begda >= @p_dtfin
      AND p0000~stat2 <> '3'
      AND p0001~bukrs IN @s_bukrs
    INTO TABLE @lt_raw.

  " 2) Una sola riga per PERNR: tengo quella con BEGDA più recente
  LOOP AT lt_raw INTO ls_raw.
    READ TABLE gt_cessati WITH TABLE KEY pernr = ls_raw-pernr INTO ls_existing.

    IF sy-subrc <> 0.
      INSERT VALUE ts_input( pernr = ls_raw-pernr
                             begda = ls_raw-begda
                             endda = ls_raw-endda ) INTO TABLE gt_cessati.
    ELSEIF ls_raw-begda > ls_existing-begda.
      ls_existing-begda = ls_raw-begda.
      ls_existing-endda = ls_raw-endda.
      MODIFY TABLE gt_cessati FROM ls_existing.
    ENDIF.
  ENDLOOP.

  " 3) Output finale (standard table)
  gt_cessati_2 = CORRESPONDING #( gt_cessati ).


ENDFORM.

FORM calcola_date_cessati.
  DATA: lv_datum_prec TYPE dats.
  " ----- Calcolo primo e ultimo giorno dei mese precedenti per i cessati -----------
  lv_datum_prec       = sy-datum.
  lv_datum_prec+6(02) = '01'.
  lv_datum_prec       = lv_datum_prec - 1.

  p_dtfin = lv_datum_prec.
  DATA mesi LIKE p_mesi.
  mesi = p_mesi - 1.
  CALL FUNCTION 'RP_CALC_DATE_IN_INTERVAL'
    EXPORTING
      date      = p_dtfin
      days      = 0
      months    = mesi
      years     = 0
      signum    = '-'
    IMPORTING
      calc_date = lv_datum_prec.

  lv_datum_prec+6(2) = '01'.
  p_dtinz = lv_datum_prec.
ENDFORM.

FORM get_data_0001 TABLES   gt_inp STRUCTURE wa_input.

  DATA: lt_pa0001_raw    TYPE STANDARD TABLE OF ts_input,
        lt_pa0001_latest TYPE HASHED TABLE OF   ts_input WITH UNIQUE KEY pernr,
        ls_pa0001        TYPE ts_input.

  FIELD-SYMBOLS: <fs_inp> TYPE ts_input.

  IF gt_inp[] IS NOT INITIAL.

    SELECT pernr begda endda persg FROM pa0001 INTO TABLE lt_pa0001_raw
      FOR ALL ENTRIES IN gt_inp
      WHERE pernr = gt_inp-pernr.

    IF sy-subrc = 0 AND lt_pa0001_raw IS NOT INITIAL.
      SORT lt_pa0001_raw BY pernr begda DESCENDING endda DESCENDING.
      LOOP AT lt_pa0001_raw INTO ls_pa0001.
        " Inserisco solo il primo record per PERNR,
        " che dopo il sort è quello più recente
        INSERT ls_pa0001 INTO TABLE lt_pa0001_latest.
        IF sy-subrc <> 0.
          CONTINUE.
        ENDIF.
      ENDLOOP.

      LOOP AT gt_inp ASSIGNING <fs_inp>.
        READ TABLE lt_pa0001_latest INTO ls_pa0001 WITH TABLE KEY pernr = <fs_inp>-pernr.
        IF sy-subrc = 0.
          <fs_inp>-begda = ls_pa0001-begda.
          <fs_inp>-endda = ls_pa0001-endda.
          <fs_inp>-persg = ls_pa0001-persg.
        ENDIF.
      ENDLOOP.

    ENDIF.

  ENDIF.

ENDFORM.
